function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Sheet2"); var folder = DriveApp.getFolderById("1fFPCGCp7Nz9CfdhAOjn0j8egzhvidP_R"); var data = JSON.parse(e.postData.contents); var namaInput = data.nama.trim(); var keterangan = data.keterangan; var durasiBreak = data.durasiBreak || ""; var linkFoto = ""; // Simpan foto ke Google Drive HANYA jika bukan Break if (data.foto && data.foto.indexOf(',') !== -1) { var bagianFoto = data.foto.split(','); var decoded = Utilities.base64Decode(bagianFoto); var blob = Utilities.newBlob(decoded, "image/jpeg", "Absen_" + namaInput + "_" + keterangan + "_" + new Date().getTime() + ".jpg"); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); linkFoto = file.getUrl(); } // Format tanggal dan jam sekarang (WIB) var tanggalSekarang = Utilities.formatDate(new Date(), "Asia/Jakarta", "dd/MM/yyyy"); var jamSekarang = Utilities.formatDate(new Date(), "Asia/Jakarta", "h:mm:ss a"); // Cari baris nama di Kolom B pada Sheet2 var lastRow = sheet.getLastRow(); var namaRange = sheet.getRange(1, 2, lastRow, 1).getValues(); var barisDitemukan = -1; for (var i = 0; i < namaRange.length; i++) { if (namaRange[i].toString().trim().toLowerCase() === namaInput.toLowerCase()) { barisDitemukan = i + 1; break; } } // Jika nama tidak ada di daftar, buat baris baru di bawah if (barisDitemukan === -1) { barisDitemukan = lastRow + 1; sheet.getRange(barisDitemukan, 2).setValue(namaInput); } // Isi tanggal di Kolom A sheet.getRange(barisDitemukan, 1).setValue(tanggalSekarang); // Isi jam dan link foto murni di kolom masing-masing if (keterangan === "Masuk") { sheet.getRange(barisDitemukan, 3).setValue(jamSekarang); if (linkFoto !== "") { sheet.getRange(barisDitemukan, 6).setValue(linkFoto); // Kolom F = FOTO MASUK } } else if (keterangan === "Break") { sheet.getRange(barisDitemukan, 4).setValue(durasiBreak + " (" + jamSekarang + ")"); } else if (keterangan === "Pulang") { sheet.getRange(barisDitemukan, 5).setValue(jamSekarang); if (linkFoto !== "") { sheet.getRange(barisDitemukan, 7).setValue(linkFoto); // Kolom G = FOTO PULANG } } return ContentService.createTextOutput(JSON.stringify({"status": "sukses"})).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()})).setMimeType(ContentService.MimeType.JSON); } }
